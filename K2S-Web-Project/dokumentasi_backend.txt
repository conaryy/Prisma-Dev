1) Skema Database (MySQL)

CREATE DATABASE IF NOT EXISTS k2s_p1
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE k2s_p1;

-- Instansi (profil)
CREATE TABLE IF NOT EXISTS instansi (
  id CHAR(36) PRIMARY KEY,
  facility_group VARCHAR(255) NOT NULL,
  kobu VARCHAR(255) NOT NULL,
  bandara VARCHAR(255) NOT NULL,
  cabang_airnav VARCHAR(255) NOT NULL,
  name VARCHAR(255) NOT NULL,
  alias VARCHAR(255) DEFAULT NULL,
  address TEXT,
  contact_name VARCHAR(255) DEFAULT NULL,
  contact_email VARCHAR(255) DEFAULT NULL,
  contact_phone VARCHAR(100) DEFAULT NULL,
  fax VARCHAR(100) DEFAULT NULL,
  postal_code VARCHAR(20) DEFAULT NULL,
  longitude VARCHAR(50) DEFAULT NULL,
  latitude VARCHAR(50) DEFAULT NULL,
  service_area VARCHAR(255) DEFAULT NULL,
  operating_hours VARCHAR(100) DEFAULT NULL,
  traffic_type VARCHAR(100) DEFAULT NULL,
  runway_capacity VARCHAR(100) DEFAULT NULL,
  runway_condition VARCHAR(100) DEFAULT NULL,
  location VARCHAR(255) DEFAULT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- Fasilitas
CREATE TABLE IF NOT EXISTS facilities (
  id CHAR(36) PRIMARY KEY,
  instansi_id CHAR(36) DEFAULT NULL,
  facility_group VARCHAR(255) NOT NULL,
  name VARCHAR(255) NOT NULL,
  merk VARCHAR(255) DEFAULT NULL,
  type VARCHAR(255) DEFAULT NULL,
  frequency VARCHAR(100) DEFAULT NULL,
  ident VARCHAR(100) DEFAULT NULL,
  condition_status ENUM('Normal','Single','Unserviceable') NOT NULL DEFAULT 'Normal',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_facilities_instansi
    FOREIGN KEY (instansi_id) REFERENCES instansi(id)
    ON DELETE SET NULL
);

-- Organisasi
CREATE TABLE IF NOT EXISTS organizations (
  id CHAR(36) PRIMARY KEY,
  instansi_id CHAR(36) DEFAULT NULL,
  unit_name VARCHAR(255) NOT NULL,
  position_name VARCHAR(255) NOT NULL,
  level VARCHAR(50) DEFAULT NULL,
  status ENUM('Aktif','Nonaktif') NOT NULL DEFAULT 'Aktif',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_organizations_instansi
    FOREIGN KEY (instansi_id) REFERENCES instansi(id)
    ON DELETE SET NULL
);

-- Personil
CREATE TABLE IF NOT EXISTS personil (
  id CHAR(36) PRIMARY KEY,
  instansi_id CHAR(36) DEFAULT NULL,
  name VARCHAR(255) NOT NULL,
  position VARCHAR(255) DEFAULT NULL,
  level VARCHAR(50) DEFAULT NULL,
  field_name VARCHAR(100) DEFAULT NULL,
  tier VARCHAR(100) DEFAULT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_personil_instansi
    FOREIGN KEY (instansi_id) REFERENCES instansi(id)
    ON DELETE SET NULL
);

-- Sertifikat
CREATE TABLE IF NOT EXISTS certificates (
  id CHAR(36) PRIMARY KEY,
  instansi_id CHAR(36) DEFAULT NULL,
  name VARCHAR(255) NOT NULL,
  type VARCHAR(255) NOT NULL,
  number VARCHAR(255) NOT NULL,
  valid_until DATE NOT NULL,
  status ENUM('Berlaku','Tidak Berlaku') NOT NULL DEFAULT 'Berlaku',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_certificates_instansi
    FOREIGN KEY (instansi_id) REFERENCES instansi(id)
    ON DELETE SET NULL
);

-- Agenda
CREATE TABLE IF NOT EXISTS agenda (
  id CHAR(36) PRIMARY KEY,
  date DATE NOT NULL,
  activity_type VARCHAR(255) NOT NULL,
  location VARCHAR(255) NOT NULL,
  status VARCHAR(100) NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- Audit
CREATE TABLE IF NOT EXISTS audits (
  id CHAR(36) PRIMARY KEY,
  number VARCHAR(255) NOT NULL,
  date DATE NOT NULL,
  location VARCHAR(255) NOT NULL,
  status VARCHAR(100) NOT NULL,
  focus VARCHAR(255) NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- Inspeksi
CREATE TABLE IF NOT EXISTS inspections (
  id CHAR(36) PRIMARY KEY,
  number VARCHAR(255) NOT NULL,
  date DATE NOT NULL,
  location VARCHAR(255) NOT NULL,
  status VARCHAR(100) NOT NULL,
  unit VARCHAR(255) NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- Monitoring
CREATE TABLE IF NOT EXISTS monitoring (
  id CHAR(36) PRIMARY KEY,
  date DATE NOT NULL,
  location VARCHAR(255) NOT NULL,
  status VARCHAR(100) NOT NULL,
  note TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- Temuan (Finding)
CREATE TABLE IF NOT EXISTS findings (
  id CHAR(36) PRIMARY KEY,
  source_type ENUM('Audit','Inspeksi','Monitoring') NOT NULL,
  source_id CHAR(36) NOT NULL,
  kategori VARCHAR(255) NOT NULL,
  deskripsi TEXT NOT NULL,
  severity ENUM('Low','Med','High') NOT NULL,
  status ENUM('Open','In Progress','Closed') NOT NULL,
  due_date DATE NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX idx_findings_source (source_type, source_id)
);

-- Tindak Lanjut (Follow Up / CAP)
CREATE TABLE IF NOT EXISTS follow_ups (
  id CHAR(36) PRIMARY KEY,
  finding_id CHAR(36) NOT NULL,
  aksi_perbaikan TEXT NOT NULL,
  pic VARCHAR(255) NOT NULL,
  target_date DATE NOT NULL,
  status ENUM('Open','In Progress','Done') NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_followups_finding
    FOREIGN KEY (finding_id) REFERENCES findings(id)
    ON DELETE CASCADE
);

-- Riwayat CAP
CREATE TABLE IF NOT EXISTS follow_up_history (
  id CHAR(36) PRIMARY KEY,
  follow_up_id CHAR(36) NOT NULL,
  note TEXT NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_followup_history
    FOREIGN KEY (follow_up_id) REFERENCES follow_ups(id)
    ON DELETE CASCADE
);

-- Evidence CAP
CREATE TABLE IF NOT EXISTS follow_up_evidence (
  id CHAR(36) PRIMARY KEY,
  follow_up_id CHAR(36) NOT NULL,
  name VARCHAR(255) NOT NULL,
  url TEXT NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_followup_evidence
    FOREIGN KEY (follow_up_id) REFERENCES follow_ups(id)
    ON DELETE CASCADE
);

2) Catatan Implementasi
- source_id pada findings tidak bisa diproteksi FK karena polymorphic; validasi dilakukan di aplikasi.
- Jika hanya 1 instansi, bisa pakai single row pada instansi dan simpan instansi_id di tabel lain.
- Anda bisa mengganti id CHAR(36) dengan BINARY(16) jika ingin lebih hemat (butuh UUID_TO_BIN).

3) Best Practice Anti-SQL Injection (di sisi backend)
- Selalu gunakan prepared statements / parameterized queries.
- Jangan membangun SQL dari string input user (terutama WHERE, ORDER BY, LIMIT, JOIN).
- Whitelist untuk field yang boleh di-sort/filter (ORDER BY harus berbasis whitelist).
- Terapkan least privilege untuk user DB (batasi hak akses l12 sesuai kebutuhan).
- Nonaktifkan multi-statement di driver jika memungkinkan.
- Logging, rate limiting, dan validasi input di level aplikasi.

8) Draft API Endpoints (Best Practice)
- Semua response JSON standar: { "data": ..., "meta": ..., "error": ... }
- Semua request yang mengubah data membutuhkan auth (Bearer token).
- Pagination: gunakan query ?page=1&limit=20; response meta: { page, limit, total, totalPages }.
- Filter dan sort wajib whitelist di backend.

9) Payload + Validasi (Laravel Sanctum)
Auth (Sanctum token)
- POST /api/auth/login
  body: { email, password, remember? }
  validate: email|required|email, password|required|min:8, remember|boolean
  resp: { data: { token, token_type: "Bearer", user } }
- POST /api/auth/logout
  header: Authorization: Bearer <token>
  resp: { data: { success: true } }
- GET /api/auth/me
  header: Authorization: Bearer <token>
  resp: { data: { id, name, email, role, status } }

Users (admin only)
- POST /api/users
  body: { name, email, password, role, status }
  validate:
    name|required|min:2
    email|required|email|unique:users
    password|required|min:8
    role|required|in:Admin,Supervisor,Auditor,Viewer
    status|in:Active,Disabled
- PATCH /api/users/:id
  body: { name?, email?, password?, role?, status? }
  validate: same as create (email unique except self)

Instansi
- PUT /api/instansi/:id
  body: { facilityGroup, kobu, bandara, cabangAirnav, name, alias?, address?, contactName?, contactEmail?, contactPhone?, fax?, postalCode?, longitude?, latitude?, serviceArea?, operatingHours?, trafficType?, runwayCapacity?, runwayCondition?, location? }
  validate: facilityGroup|required, kobu|required, bandara|required, cabangAirnav|required, name|required, contactEmail|email

Facilities
- POST /api/facilities
  body: { instansiId?, group, name, merk?, type?, frequency?, ident?, condition }
  validate: group|required, name|required, condition|in:Normal,Single,Unserviceable
- PATCH /api/facilities/:id
  body: same fields optional

Organizations
- POST /api/organizations
  body: { instansiId?, unitName, positionName, level?, status }
  validate: unitName|required, positionName|required, status|in:Aktif,Nonaktif

Personil
- POST /api/personil

Certificates

  body: { instansiId?, name, type, number, validUntil, status }
  validate: name|required, type|required, number|required, validUntil|required|date, status|in:Berlaku,Tidak Berlaku

Agenda
- POST /api/agenda

  validate: date|required|date, activityType|required, location|required, status|required

Audits
- POST /api/audits

Inspections
- POST /api/inspections
